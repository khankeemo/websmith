SELECT cm.id AS message_id, cm.conversation_id, cm.sender_name, cm.sender_email, cm.message, cm.created_at,
cc.subject AS conv_subject, cc.mailbox_id, mb.email_address AS mailbox_email, cc.customer_email
FROM conversation_messages cm
JOIN communication_conversations cc ON cc.id = cm.conversation_id
LEFT JOIN mailboxes mb ON mb.id = cc.mailbox_id
WHERE cm.sender_type='customer' AND cm.is_internal IS NOT TRUE AND cc.deleted_at IS NULL
ORDER BY cm.id DESC LIMIT 200; -- BRIDGE_BATCH_LIMIT
